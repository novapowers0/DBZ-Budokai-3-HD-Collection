cmake_minimum_required(VERSION 3.25)
project(dbz3_gpu_lab LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "The experimental EU host requires Windows and Clang with the MSVC ABI")
endif()
set(CMAKE_CXX_STANDARD 23)
find_package(rexglue REQUIRED CONFIG)
get_filename_component(DBZ3_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(DBZ3_GENERATED_EU_DIR "${DBZ3_ROOT}/generated_eu" CACHE PATH "Locally regenerated EU source directory")
if(NOT EXISTS "${DBZ3_GENERATED_EU_DIR}/sources.cmake")
  message(FATAL_ERROR "Generate EU guest code locally first; see native_gpu/INTEGRATION.md")
endif()
include("${DBZ3_GENERATED_EU_DIR}/sources.cmake")
add_executable(dbz3-gpu-lab WIN32 main.cpp trace.cpp eu_dispatch.cpp graphics_hooks.cpp native_loading.cpp native_indexed.cpp native_stream.cpp native_services.cpp reference_capture.cpp ${GENERATED_SOURCES}
  "${DBZ3_ROOT}/patches/rexglue-sdk/src/ui/rex_app.cpp"
  "${REXGLUE_SHARE_DIR}/windowed_app_main_sdl.cpp")
target_include_directories(dbz3-gpu-lab PRIVATE "${DBZ3_GENERATED_EU_DIR}")
target_link_libraries(dbz3-gpu-lab PRIVATE rex::runtime dbghelp)
option(DBZ3_REFERENCE_CAPTURE "Enable optional RenderDoc research capture" OFF)
if(DBZ3_REFERENCE_CAPTURE)
  find_path(RENDERDOC_APP_INCLUDE_DIR renderdoc_app.h REQUIRED)
  target_include_directories(dbz3-gpu-lab PRIVATE "${RENDERDOC_APP_INCLUDE_DIR}")
  target_compile_definitions(dbz3-gpu-lab PRIVATE DBZ3_REFERENCE_CAPTURE=1)
endif()
target_compile_definitions(dbz3-gpu-lab PRIVATE REXGLUE_BUILD_CONFIG="Release" _CRT_SECURE_NO_WARNINGS DBZ3_NATIVE_LAB_TRACE=1)
target_compile_options(dbz3-gpu-lab PRIVATE -msse4.1 -Wno-deprecated-declarations)
target_link_options(dbz3-gpu-lab PRIVATE -Xlinker /MAP:lab.map)
# Match the generated guest's original console semantics, including IEEE NaNs.
set_source_files_properties(${GENERATED_SOURCES} PROPERTIES COMPILE_OPTIONS "-fno-strict-aliasing;-fwrapv;-ffp-contract=off")
configure_file(lab-config.toml "${CMAKE_CURRENT_BINARY_DIR}/lab-config.toml" COPYONLY)
option(DBZ3_NATIVE_CONTRACT_TESTS "Build CPU notification and EU dispatch tests" OFF)
if(DBZ3_NATIVE_CONTRACT_TESTS)
  enable_testing()
  add_executable(native-memory-watch-test test_memory_watch.cpp)
  target_link_libraries(native-memory-watch-test PRIVATE rex::runtime)
  add_test(NAME native-memory-watch COMMAND native-memory-watch-test)
  add_executable(native-eu-dispatch-test test_dispatch.cpp eu_dispatch.cpp)
  target_include_directories(native-eu-dispatch-test PRIVATE "${DBZ3_GENERATED_EU_DIR}")
  target_link_libraries(native-eu-dispatch-test PRIVATE rex::runtime)
  add_test(NAME native-eu-dispatch COMMAND native-eu-dispatch-test)
endif()
